\section{Channel, decoder, metrics}
\paragraph{Channel.} The all-zero codeword is sent with BPSK (bit $0\mapsto +1$) over AWGN. For a point $E_b/N_0$ (dB),
\[ \sigma^2=\frac{1}{2R\,10^{(E_b/N_0)/10}},\qquad y_v=1+\sigma w_v,\ w_v\sim\mathcal N(0,1),\qquad \Lch_v=\frac{2y_v}{\sigma^2},\quad v=1,\dots,n, \]
with $R=1-m/n$. All $n$ bits are transmitted (no puncturing). One frame draws $w_1,\dots,w_n$ with \texttt{randn(st,1,n)}.

\paragraph{Sequential (check-node) BP.} Posteriors $L_v$ start at $\Lch_v$; check-to-bit messages $r_{c,v}$ start at 0.
Scheduling check $c$ with neighbours $\mathcal N(c)$ (ascending) performs, for every $v\in\mathcal N(c)$ (all $q_v$ are formed before any $L_v$ changes):
\begin{align*}
q_v&=L_v-r_{c,v},\\
t_v&=\tanh(q_v/2),\quad |t_v|\ \text{floored at }10^{-12}\ (\text{sign kept; }t_v=0\mapsto+10^{-12}),\\
p_v&=\Big(\textstyle\prod_{u\in\mathcal N(c)}t_u\Big)\big/\,t_v,\quad\text{clipped to }\big[-(1-10^{-15}),\,1-10^{-15}\big],\\
r'_{c,v}&=\ln\frac{1+p_v}{1-p_v},\qquad L_v\leftarrow q_v+r'_{c,v},\qquad r_{c,v}\leftarrow r'_{c,v}.
\end{align*}
Scheduling a \emph{cluster} applies this to its checks one after another in increasing index order.

\paragraph{Iterations and stopping.} One iteration schedules every cluster exactly once; the policy only chooses the order.
After each iteration the syndrome of $\hat x_v=\mathbb 1[L_v<0]$ is checked and decoding stops when it is zero. At most
$\ell_{\max}$ iterations ($\ell_{\max}=5$ for WRAN, $50$ for BG2~Z=10), same value in training and evaluation.

\paragraph{Flooding.} Per iteration all checks update from the previous iteration's bit-to-check messages
(same $t$/$p$/$\ln$ formulas and clamps), then $L_v=\Lch_v+\sum_c r_{c,v}$ and bit-to-check messages $L_v-r_{c,v}$;
syndrome of $\mathbb 1[L_v\le 0]$ checked after every iteration.

\paragraph{Metrics.} Over $F$ frames of $n$ bits: $\mathrm{BER}=\sum(\text{bits with }L_v<0)/(Fn)$ (flooding: $L_v\le 0$),
$\mathrm{FER}=\#\{\text{frames with any bit error}\}/F$, average iterations $=\text{messages}/(F\cdot\mathrm{nnz}(H))$ where scheduling a check adds its degree to \emph{messages} (flooding adds $\mathrm{nnz}(H)$ per iteration).

\section{Clusters, states, rewards, policies}
\paragraph{Clusters.} $C_k$, $k=1..K$, are contiguous groups of $z$ checks ($K=\lceil m/z\rceil$; the last group may be smaller). The neighbourhood $\mathcal N_k$ is the sorted union of its checks' bits: $v_{k,1}<\dots<v_{k,|\mathcal N_k|}$. WRAN: $|\mathcal N_k|\in\{10,11\}$ ($z=1$), up to 44 ($z=4$), 160--176 ($z=16$).

\paragraph{Round-robin (layered).} Order $1,2,\dots,K$ every iteration.

\subsection{RELDEC (binary-vector state)}
State of cluster $k$: $b_{k,i}=\mathbb 1[L_{v_{k,i}}>0]$, $i=1..|\mathcal N_k|$. Reward after scheduling $a$:
$\displaystyle R=\frac{1}{|\mathcal N_a|}\sum_{i}\mathbb 1[L_{v_{a,i}}>0]$ (evaluated after the update).
Q storage (implementation only; values are identical): integer key $\sum_i b_{k,i}2^{i-1}$, dense $2^{|\mathcal N|}\times K$ table if the largest neighbourhood has $\le 20$ bits ($z=1$), otherwise a list of visited keys per cluster ($\le 52$ bits, $z=4$), otherwise one hash map per cluster keyed by the bit string ($z=16$). Unvisited states have $Q=0$.

\subsection{tanh-MI (averaged soft state, MI reward)}
State: $\displaystyle x_k=\frac{1}{|\mathcal N_k|}\sum_{v\in\mathcal N_k}\tanh(L_v)$ (note: $\tanh(L_v)$, not $\tanh(L_v/2)$; $L_v$ unclipped), $\ s_k=\big\lfloor 10\,(\mathrm{clip}(x_k,-1,1)+1)+0.5\big\rfloor\in\{0,\dots,20\}$.
Reward after scheduling $a$: $\displaystyle R=\frac{1}{|\mathcal N_a|}\sum_{v\in\mathcal N_a}\big[I(L_v^{\text{after}})-I(L_v^{\text{before}})\big]$, with
\[ s=\sqrt{\max(2L,0)},\qquad I(L)=\mathrm{clip}_{[0,1]}\begin{cases}-0.0421061\,s^3+0.209252\,s^2-0.00640081\,s, & s\le 1.6363,\\ 1-\exp(0.00181491\,s^3-0.142675\,s^2-0.0822054\,s+0.0549608), & 1.6363<s<10,\\ 1, & s\ge 10.\end{cases} \]
Q table: $21\times K$, initialised to 0.

\subsection{Tabular Q-learning (RELDEC and tanh-MI)}
Each cluster is a one-action sub-MDP with value $Q_k(s)$. After scheduling cluster $a$ (state $s$ just before, $s'$ just after, both of cluster $a$):
\[ Q_a(s)\leftarrow Q_a(s)+\alpha\big(R+\gamma\,Q_a(s')-Q_a(s)\big),\qquad \alpha=0.1,\ \gamma=0.99. \]
No maximum over other clusters. Selection among clusters not yet scheduled in the current iteration: training is $\varepsilon$-greedy ($\varepsilon=0.1$: with probability $\varepsilon$ a uniformly random available cluster, otherwise $\arg\max_k Q_k(s_k)$); evaluation is greedy. Ties in the arg\,max are broken uniformly at random (\texttt{randi} over the tied set).

\subsection{PPO}
\textbf{Observation} (\texttt{ppo\_features.m}), one row per cluster $k$, computed before each decision:
\begin{itemize}
\item $f_{k,1}=\frac{1}{|\mathcal N_k|}\sum_{v\in\mathcal N_k}\tanh(\mathrm{clip}(L_v,-30,30))$;
\item $f_{k,2}$ = fraction of $C_k$'s checks unsatisfied under $\mathbb 1[L_v<0]$;
\item $f_{k,3}=\min_{v\in\mathcal N_k}|\tanh(L_v/2)|$;
\item $f_{k,4},f_{k,5}$ = $f_{\cdot,1}$ and $f_{\cdot,3}$ z-scored across all $K$ clusters: $(f-\bar f)/(\mathrm{std}_{\text{pop}}(f)+10^{-6})$;
\item $f_{k,6}=\mathbb 1[k\text{ still available this iteration}]$; $f_{k,7}=(k-1)/(K-1)$ only if \emph{pos\_feature};
\item three global features appended to every row: $(\ell-1)/\ell_{\max}$, $(j-1)/K$ ($j$ = step within the iteration), fraction of all checks unsatisfied. Input width $D=F+3$ ($F=6$ or 7).
\end{itemize}

\textbf{Network} (hidden width $h=32$, all activations $\tanh$): actor $s_k=W_3\tanh(W_2\tanh(W_1x_k+b_1)+b_2)+b_3$ shared over clusters,
$\pi(a)=\mathrm{softmax}$ of $s$ over available clusters; critic $e_k=\tanh(E_2\tanh(E_1x_k+c_1)+c_2)$,
$V=C_2\tanh(C_1[\bar e;g]+d_1)+d_2$ with $\bar e$ the mean over all $K$ clusters and $g$ the 3 global features.
Initialisation (\texttt{ppo\_init.m}): every weight and bias uniform in $\pm1/\sqrt{\text{fan-in}}$.

\textbf{Acting.} Training samples $a=\min\{k:\sum_{i\le k}\pi_i\ge u\}$, $u\sim U(0,1)$; evaluation takes $\arg\max_k s_k$ (first index on ties). Training checks the syndrome after every step and ends the episode on success.

\textbf{Reward} (both codes used \emph{success}): $R_t=1$ on the step the syndrome becomes zero, else 0.

\textbf{Update} (\texttt{ppo\_update.m}), run at the end of an episode once the buffer holds at least \emph{rollout\_steps} steps ($d_t=1$ on terminal steps, so no bootstrap is needed):
\begin{align*}
\delta_t&=R_t+\gamma V_{t+1}(1-d_t)-V_t, \qquad \hat A_t=\delta_t+\gamma\lambda(1-d_t)\hat A_{t+1}, \qquad \hat R_t=\hat A_t+V_t,\\
\tilde A_t&=(\hat A_t-\overline{\hat A})/(\mathrm{std}(\hat A)+10^{-8})\quad(\text{unbiased std, over the whole rollout}),\\
\mathcal L&=-\overline{\min\!\big(\rho_t\tilde A_t,\ \mathrm{clip}(\rho_t,1-\epsilon,1+\epsilon)\,\tilde A_t\big)}+c_v\,\overline{(V_t-\hat R_t)^2}-c_e\,\overline{H(\pi_t)},\qquad \rho_t=\exp(\log\pi(a_t)-\log\pi_{\text{old}}(a_t)),
\end{align*}
averaged over a minibatch; \emph{update\_epochs} passes over a fresh random permutation each; global gradient norm clipped to \emph{max\_grad\_norm}; Adam with $\beta_1=0.9$, $\beta_2=0.999$, $\epsilon_{\text{Adam}}=10^{-8}$, bias-corrected, learning rate \emph{lr}. $\lambda$ = \emph{gae\_lambda}, $\epsilon$ = \emph{clip\_eps}, $c_v$ = \emph{vf\_coef}, $c_e$ = \emph{ent\_coef}.

\section{Training protocol and random numbers}\label{sec:train}
\texttt{train/sched\_train.m}: episode $e=1,\dots,E$ is one fresh frame at training SNR \texttt{train\_snrs}$[(e-1)\bmod |\texttt{train\_snrs}|]$ (cycled in order),
decoded with $\ell_{\max}$ iterations under training rules. Training SNRs (dB): WRAN \{1, 1.5, 2, 2.5, 3\}; BG2 Z=10 \{1, 1.5, 2, 2.5, 3, 3.5\}.
\textbf{Seeds} (all learned policies use \texttt{seed} $=42$). Noise comes from its own stream, everything else (exploration,
tie-breaks, PPO sampling, PPO initialisation, minibatch shuffling) from the global stream, set before the policy is created:
\begin{verbatim}
st = RandStream('Threefry', 'Seed', 42);          % noise: randn(st, 1, n) per episode
RandStream.setGlobalStream(RandStream('Threefry', 'Seed', 42 + 7919));
\end{verbatim}
Under Octave (Section~\ref{sec:dev}): \texttt{randn('state',42)} for noise and \texttt{rand('state',42+7919)} for the rest.
MATLAB threads: 1 (\texttt{maxNumCompThreads(1)}). Checkpoints are saved with \texttt{save(...,'-v7')} after \texttt{sched\_policy\_pack}.

\section{Evaluation protocol and random numbers}
\texttt{parallel\_eval/run\_sched\_eval.m}: for the SNR list of the stage (in the order given), each SNR is split into chunks
of 1000 frames; chunks are numbered with one global counter $k=1,2,\dots$ in that order (SNR-major). Chunk $k$ draws all its frames from
\texttt{st = RandStream('Threefry','Seed',base\_seed); st.Substream = k} (frame noise \texttt{randn(st,1,n)}, sequentially), so
the frames do not depend on the number of workers or on which worker ran the chunk. 100000 frames per SNR (no early stop),
greedy policy, same $\ell_{\max}$ as training. The \texttt{base\_seed} of every run is in Table~\ref{tab:runs}.
Tie-breaking draws (\texttt{rand}/\texttt{randi} inside the policy) come from each parallel worker's default global stream and are
\emph{not} pinned to the chunk (Section~\ref{sec:dev}).

\section{How to reproduce}\label{sec:howto}
From \texttt{external\_ref/ldpc\_rl/} in MATLAB:
\begin{verbatim}
addpath('lib'); addpath('train'); addpath('parallel_eval');
% 1) train (example: tanh-MI, WRAN, z = 1, 1000 episodes)
cfg = struct('code','wran', 'z',1, 'method','tanh_mi', 'episodes',1000, ...
             'l_max',5, 'train_snrs',[1 1.5 2 2.5 3], 'seed',42, ...
             'alpha',0.1, 'gamma',0.99, 'epsilon',0.1, ...
             'out','my_tanh_z1_ep1000.mat');
ck = sched_train(cfg);
% PPO: method 'ppo' plus a field cfg.ppo holding the Section 4.4 values.
% 2) evaluate: write a JSON file with fields
%    manifest   struct: method ('matlab_...'), target_frame_errors, max_frames
%    code, z, kind ('flooding'|'round_robin'|'reldec'|'tanh_mi'|'ppo')
%    checkpoint (learned kinds), snr_db, n_frames (100000), chunk_size (1000)
%    i_max, base_seed (Section 7), n_workers, spool_dir
%  then
run_sched_eval('eval_cfg.json');
\end{verbatim}
Each chunk is written to \texttt{spool\_dir/<kind>\_snr<SNR>\_c<index>.json} with \texttt{frames}, \texttt{bit\_errors},
\texttt{total\_bits}, \texttt{frame\_errors}, \texttt{messages}, \texttt{iterations}; sum them per SNR and apply the metric definitions
of Section~3. To use the exact trained policies, point \texttt{checkpoint} at the files in \texttt{checkpoints/sched/}.
The exact JSON written for every run is produced by \texttt{matlab\_bridge/sched\_driver.py} (\texttt{run\_stage}).

\section{Deviations and reproducibility limits}\label{sec:dev}
\begin{itemize}
\item \textbf{Octave.} Ada's MATLAB license server failed (error $-97$) during part of the work. GNU Octave 9.2
(container \texttt{docker://gnuoctave/octave:9.2.0}) ran the same \texttt{.m} files for: WRAN flooding (evaluation), and the training
of WRAN RELDEC $z{=}1,4$ and tanh-MI $z{=}1,4$ at 500 episodes; 61 of the 600 WRAN RELDEC $z{=}1$ evaluation chunks were also run in Octave.
Octave lacks \texttt{RandStream}: it seeds training with \texttt{randn/rand('state',\dots)} (Section~\ref{sec:train}) and evaluation
chunk $k$ with \texttt{randn('state', base\_seed$\cdot$100003$+k$)}. Those Octave checkpoints were re-saved in MATLAB format
(\texttt{matlab\_bridge/ada\_jobs/convert\_ckpt.m}); values unchanged.
\item \textbf{Bit-exactness.} Flooding, round-robin and PPO (greedy, deterministic) are bit-reproducible from the seeds.
RELDEC and tanh-MI evaluations break Q ties with worker-local random streams, so individual frames can differ between runs;
the statistics are reproducible within Monte-Carlo error (as for the Python implementation, whose tie-breaking is unseeded).
\item \textbf{Retries.} Two BG2 Z=10 15k-episode stages failed once (a parallel-pool worker crashed at start-up) and were rerun;
chunk results do not depend on worker count, so partially completed sweeps were resumed.
\item \textbf{Cancelled.} BG2 PPO with 10\,000 episodes was not completed; it has no results.
\item \textbf{PPO on WRAN.} MATLAB PPO $z{=}1$ has FER 0.416 at 2~dB versus 0.221 for the Python PPO run. The PPO update and features
were verified equal to the Python implementation (to float32 rounding); the difference is single-seed training variance.
\end{itemize}
